\section{Phase 3: Community, Notifications \& Admin Portal}

\subsection{Scope Overview}

Phase 3 focuses on building the social and administrative backbone of the TeeTime Golf App.

\begin{tabularx}{\textwidth}{|c|Y|}
\hline
\rowcolor{phase3!20}
\textbf{\#} & \textbf{Deliverable} \\
\hline
1 & Community Section (QR scan to add friends, Social Rate system) \\
\hline
2 & Waiver Wire (availability posts and notifications) \\
\hline
3 & Firebase Cloud Messaging (FCM) for push notifications \\
\hline
4 & Super Admin Web Portal (Dashboard, Users, Courses, Clubs, Sponsors, Notifications) \\
\hline
\end{tabularx}


\subsection{Feature Requirements}

\subsubsection{Community Module}

\begin{tabularx}{\textwidth}{|p{3cm}|Y|}
\hline
\rowcolor{lightgray}
\textbf{Feature} & \textbf{Description} \\
\hline
QR Code Scanning & Users can scan another golfer's QR code to add them to their roster/On-Deck list \\
\hline
Friend Discovery & Search and discover golfers by name, location, or club membership \\
\hline
Social Rate Display & Show Social Rate score (0-150+) on user profiles with history \\
\hline
Mutual Connections & Indicate when two users have added each other \\
\hline
Block/Report & Allow users to block or report inappropriate behavior \\
\hline
\end{tabularx}

\subsubsection{Waiver Wire System}

\begin{tabularx}{\textwidth}{|p{3cm}|Y|}
\hline
\rowcolor{lightgray}
\textbf{Feature} & \textbf{Description} \\
\hline
Availability Posts & Users post when they're available for a round (date, time, location preferences) \\
\hline
Feed Display & Chronological feed of availability posts from roster members \\
\hline
Quick Invite & One-tap invite from waiver wire post to create tee time \\
\hline
Expiration & Posts auto-expire after the specified date/time passes \\
\hline
Notifications & Push notification when roster member posts availability \\
\hline
\end{tabularx}

\subsubsection{Push Notifications (FCM)}

\begin{tabularx}{\textwidth}{|p{4cm}|Y|}
\hline
\rowcolor{lightgray}
\textbf{Notification Type} & \textbf{Trigger} \\
\hline
Tee Time Invite & When user receives invitation to a tee time \\
\hline
Invite Response & When invitee accepts/declines invitation \\
\hline
Roster Add & When someone adds user to their roster \\
\hline
Waiver Wire & When roster member posts availability \\
\hline
Reminder & 24hr and 2hr before confirmed tee time \\
\hline
Social Rate Change & When Social Rate score changes significantly \\
\hline
Chat Message & New message in group chat (Phase 4) \\
\hline
\end{tabularx}

\subsubsection{Super Admin Web Portal}

\begin{tabularx}{\textwidth}{|p{3cm}|Y|}
\hline
\rowcolor{lightgray}
\textbf{Module} & \textbf{Functionality} \\
\hline
Dashboard & Analytics overview: active users, tee times created, app engagement metrics \\
\hline
User Management & View/search users, suspend accounts, reset passwords, view activity logs \\
\hline
Courses & Add/edit/delete golf courses, manage course details and images \\
\hline
Clubs & Manage club memberships, verify club affiliations \\
\hline
Sponsors & Create sponsor profiles, manage sponsorship containers/ads \\
\hline
Notifications & Send broadcast notifications, manage notification templates \\
\hline
Reports & Generate usage reports, export data \\
\hline
\end{tabularx}


\subsection{Phase 3 Deliverables}

\begin{enumerate}[label=\textbf{\arabic*.}]
\item Functional community module integrated in mobile app
\item Waiver wire feature with real-time updates
\item Push notification system via Firebase FCM
\item Deployed admin portal connected to backend API
\item Admin portal documentation
\end{enumerate}

\subsection{Firebase Credentials (Phase 3)}

\begin{tabularx}{\textwidth}{|p{4cm}|p{3cm}|Y|}
\hline
\rowcolor{warning!30}
\textbf{Credential} & \textbf{Format} & \textbf{Purpose} \\
\hline
\multicolumn{3}{|l|}{\cellcolor{lightblue}\textbf{Firebase Project Configuration}} \\
\hline
Project ID & String & Identifies your Firebase project (e.g., ``teetime-golf-prod'') \\
\hline
API Key & String & Client-side API key for Firebase SDK initialization \\
\hline
Auth Domain & URL & Firebase authentication domain \\
\hline
Storage Bucket & URL & Cloud Storage bucket for file uploads \\
\hline
Messaging Sender ID & Number & FCM sender identification \\
\hline
App ID & String & Firebase app identifier \\
\hline
\multicolumn{3}{|l|}{\cellcolor{lightblue}\textbf{Firebase Cloud Messaging (FCM)}} \\
\hline
Server Key & String & Backend server key for sending push notifications \\
\hline
FCM API Key & String & Legacy server key (if using HTTP v1 API) \\
\hline
Service Account JSON & JSON File & Private key file for Firebase Admin SDK \\
\hline
\end{tabularx}

\vspace{0.3cm}
\textbf{How to Obtain Firebase Credentials:}
\begin{enumerate}[label=\arabic*.]
\item Go to \href{https://console.firebase.google.com}{Firebase Console}
\item Create a new project or select existing project
\item Navigate to \textbf{Project Settings} (gear icon)
\item Under \textbf{General} tab, find your project configuration
\item Under \textbf{Cloud Messaging} tab, find Server Key
\item Under \textbf{Service Accounts} tab, click ``Generate new private key'' for Admin SDK
\end{enumerate}

\vspace{0.3cm}
\textbf{Why We Need These:}
\begin{itemize}[label=\faCheckCircle]
\item \textbf{Project Config}: Initialize Firebase SDK in mobile app
\item \textbf{Server Key}: Send push notifications from backend server
\item \textbf{Service Account}: Authenticate backend with Firebase Admin privileges
\end{itemize}


\subsection{Twilio Credentials (Phase 4)}

\begin{tabularx}{\textwidth}{|p{4cm}|p{3cm}|Y|}
\hline
\rowcolor{warning!30}
\textbf{Credential} & \textbf{Format} & \textbf{Purpose} \\
\hline
Account SID & String (34 chars) & Unique identifier for your Twilio account \\
\hline
Auth Token & String & Secret token for API authentication \\
\hline
Phone Number & +1XXXXXXXXXX & Twilio phone number for sending SMS \\
\hline
Messaging Service SID & String & (Optional) For high-volume messaging \\
\hline
\end{tabularx}

\vspace{0.3cm}
\textbf{How to Obtain Twilio Credentials:}
\begin{enumerate}[label=\arabic*.]
\item Sign up at \href{https://www.twilio.com}{twilio.com}
\item Navigate to \textbf{Console Dashboard}
\item Find \textbf{Account SID} and \textbf{Auth Token} on the dashboard
\item Go to \textbf{Phone Numbers} → \textbf{Buy a Number}
\item Purchase a number with SMS capability
\item (Optional) Create a Messaging Service under \textbf{Messaging} → \textbf{Services}
\end{enumerate}

\vspace{0.3cm}
\textbf{Why We Need These:}
\begin{itemize}[label=\faCheckCircle]
\item \textbf{Account SID}: Identify your Twilio account in API calls
\item \textbf{Auth Token}: Authenticate API requests securely
\item \textbf{Phone Number}: Send SMS messages to tee time participants
\item \textbf{Messaging Service}: Handle high-volume SMS with automatic number pooling
\end{itemize}

\vspace{0.3cm}
\textbf{Estimated Costs:}
\begin{itemize}
\item Phone Number: \$1.00/month
\item Outbound SMS (US): \$0.0079/message
\item Inbound SMS (US): \$0.0075/message
\end{itemize}

\subsubsection{Apple Developer Account}

\begin{tabularx}{\textwidth}{|p{4cm}|p{3cm}|Y|}

\rowcolor{warning!30}

\hline

\hline
Team ID & String (10 chars) & Identifies your development team \\
\hline
Bundle ID & String & Unique app identifier (e.g., com.teetime.golf) \\
\hline
App-Specific Password & String & For CI/CD automated uploads \\

Distribution Certificate & .p12 file & Code signing for App Store builds \\
\hline
Provisioning Profile & .mobileprovision & Links certificate to app and devices \\

\vspace{0.3cm}
\textbf{How to Obtain Apple Credentials:}
\begin{enumerate}[label=\arabic*.]
\item Enroll at \href{https://developer.apple.com/programs/}{Apple Developer Program} (\$99/year)
\item Log into \href{https://appstoreconnect.apple.com}{App Store Connect}
\item Create a new app with unique Bundle ID
\item In \textbf{Certificates, Identifiers \& Profiles}, create Distribution Certificate
\item Create App Store Provisioning Profile linked to your app
\item Generate App-Specific Password at \href{https://appleid.apple.com}{appleid.apple.com}
\end{enumerate}

\vspace{0.3cm}
\textbf{Why We Need These:}
\begin{itemize}[label=\faCheckCircle]
\item \textbf{Apple ID}: Access App Store Connect for app management
\item \textbf{Team ID}: Associate builds with your organization
\item \textbf{Bundle ID}: Uniquely identify your app on iOS
\item \textbf{Certificates}: Digitally sign app for App Store distribution
\end{itemize}
