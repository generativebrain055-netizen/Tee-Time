\section{Bugs Fixed}
%--------------------------------------------------

\begin{enumerate}
    \item \textbf{Group visibility} -- Non-owners couldn't see groups they were members of
    \item \textbf{Manual invite API error} -- "manual" wasn't in allowed invite types
    \item \textbf{Player ID handling} -- Roster entry IDs were being sent as user IDs
    \item \textbf{Group member selection} -- Phone numbers weren't passed for non-app users
    \item \textbf{Stale state in async} -- Used \texttt{useRef} to fix closure issues
    \item \textbf{Calendar day alignment} -- Monday-first format caused wrong day selection
    \item \textbf{Calendar date range} -- Only showed tee times for current month
\end{enumerate}

\section{Technical Improvements}
%--------------------------------------------------

\begin{itemize}[label=\done]
    \item Using \texttt{FlashList} for all lists
    \item Components memoized with \texttt{React.memo()}
    \item Callbacks using \texttt{useCallback}
    \item Using \texttt{useRef} for immediate state access
    \item TypeScript interfaces for all data types
    \item Proper error handling with detailed messages
\end{itemize}
